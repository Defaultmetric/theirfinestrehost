### Part cabinet (v2): public and internal triggers of the War Cabinet.

# Public: a War Cabinet seat is free.
ENG_cabinet_has_free_seat = {
	check_variable = {
		ENG_cabinet_seats_used < ENG_cabinet_seats_max
	}
}

# Internal: the lane focus Ernest Bevin is working on has run its 100/120/140 days.
ENG_cabinet_bevin_work_is_due = {
	has_country_flag = ENG_cabinet_bevin_working
	OR = {
		AND = {
			check_variable = {
				ENG_cabinet_bevin_stage = 1
			}
			check_variable = {
				ENG_cabinet_bevin_days > 99
			}
		}
		AND = {
			check_variable = {
				ENG_cabinet_bevin_stage = 2
			}
			check_variable = {
				ENG_cabinet_bevin_days > 119
			}
		}
		AND = {
			check_variable = {
				ENG_cabinet_bevin_stage = 3
			}
			check_variable = {
				ENG_cabinet_bevin_days > 139
			}
		}
	}
}
