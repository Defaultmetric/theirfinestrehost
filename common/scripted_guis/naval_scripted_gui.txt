scripted_gui = {
	custom_naval_invasion_warning = {
		context_type = player_context
		 
		window_name = "custom_naval_invasion_warning"	
		
		visible = {  
			has_country_flag = ui_show_invasion_warning
		}
		
		effects = {
			ok_button_click = {
				clr_country_flag = ui_show_invasion_warning
			}
			disable_button_click = {
				set_country_flag = notif_disable_inv_warning
				clr_country_flag = ui_show_invasion_warning
			}
			show_button_click = {
				# Use this recipient's saved state, not a shared global event target.
				var:ui_invaded_state = { goto_state = THIS }
				clr_country_flag = ui_show_invasion_warning
			}
		}
		
		properties = {
			sender_flag = {
				image = "[?ui_invader_country.GetFlag]"
			}
			receiver_flag = {
				image = "[This.GetFlag]"
			}
		}
		
	}

	custom_paradrop_warning = {
		context_type = player_context
		 
		window_name = "custom_paradrop_warning"	
		
		visible = {  
			has_country_flag = ui_show_paradrop_warning
			NOT = { has_country_flag = ui_show_invasion_warning }
		}
		
		effects = {
			ok_button_click = {
				clr_country_flag = ui_show_paradrop_warning
			}
			disable_button_click = {
				set_country_flag = notif_disable_para_warning
				clr_country_flag = ui_show_paradrop_warning
			}
			show_button_click = {
				var:ui_paradrop_state = { goto_state = THIS }
				clr_country_flag = ui_show_paradrop_warning
			}
		}
		
		properties = {
			sender_flag = {
				image = "[?ui_paradrop_invader_country.GetFlag]"
			}
			receiver_flag = {
				image = "[This.GetFlag]"
			}
		}
		
	}
}
