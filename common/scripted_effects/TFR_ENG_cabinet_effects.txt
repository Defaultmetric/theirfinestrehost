### Part cabinet (v2): War Cabinet seats, automatic minister lanes, Menzies's path, deferred industrial payloads.
### State per minister <k>: flags ENG_cabinet_<k>_seated (permanent), ENG_cabinet_<k>_working (a lane focus is in work),
### ENG_cabinet_<k>_lane_done, ENG_cabinet_<k>_event_pending (5-day guard); variables ENG_cabinet_<k>_stage
### (1..3 = focus in work) and ENG_cabinet_<k>_days (progress; +1 per daily tick, not while capitulated).
### Menzies: ENG_menzies_path_sigint / ENG_menzies_path_soe (set by event tfr_eng_cabinet.80), guard ENG_cabinet_menzies_choice_pending.
### Deferred payloads: flag ENG_cabinet_due_<name>, paid by on_daily_ENG (on_actions file) from 1941.1.1.
### Definitions are ordered callee first.
### Driven once a day by on_daily_ENG -> ENG_cabinet_daily_tick (TFR_ENG_cabinet_on_actions.txt).

ENG_cabinet_bevin_lane_tick = {
	if = {
		limit = {
			has_country_flag = ENG_cabinet_bevin_working
		}
		add_to_variable = {
			ENG_cabinet_bevin_days = 1
		}
	}
	if = {
		limit = {
			has_completed_focus = ENG_ernest_bevin_focus
			NOT = {
				has_country_flag = ENG_cabinet_bevin_seated
			}
		}
		ENG_cabinet_seat_bevin = yes
	}
	if = {
		limit = {
			has_country_flag = ENG_cabinet_bevin_seated
			NOT = {
				has_country_flag = ENG_cabinet_bevin_lane_done
			}
			NOT = {
				has_country_flag = ENG_cabinet_bevin_event_pending
			}
		}
		if = {
			limit = {
				NOT = {
					has_country_flag = ENG_cabinet_bevin_working
				}
			}
			set_country_flag = {
				flag = ENG_cabinet_bevin_event_pending
				days = 5
				value = 1
			}
			country_event = tfr_eng_cabinet.11
		}
		else_if = {
			limit = {
				ENG_cabinet_bevin_work_is_due = yes
			}
			set_country_flag = {
				flag = ENG_cabinet_bevin_event_pending
				days = 5
				value = 1
			}
			country_event = tfr_eng_cabinet.12
		}
	}
}
