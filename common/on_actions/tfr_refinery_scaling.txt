on_actions = {
    on_startup = {
        effect = {
            TFR_refinery_scaling_manager_daily = yes
        }
    }
    on_daily = {
        effect = {
            if = {
                limit = {
                    OR = {
                        tag = GER
                        AND = {
                            NOT = { country_exists = GER }
                            tag = ENG
                        }
                    }
                }
                TFR_refinery_scaling_manager_daily = yes
            }
        }
    }
}
