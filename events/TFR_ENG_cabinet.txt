### Part cabinet (v2): events of the War Cabinet.
### .<i>1 = start work on the next lane focus, .<i>2 = complete the lane focus in work (hidden; i = 1 bevin, 2 cripps,
### 3 beaverbrook, 4 menzies, 5 bracken, 6 eden, 7 bridges). Every event re-checks the state, so a stray or duplicate
### firing does nothing. .80 = Menzies's path (visible, two options).

add_namespace = tfr_eng_cabinet

# Ernest Bevin - start work
country_event = {
	id = tfr_eng_cabinet.11
	hidden = yes
	is_triggered_only = yes
	immediate = {
		clr_country_flag = ENG_cabinet_bevin_event_pending
		if = {
			limit = {
				has_country_flag = ENG_cabinet_bevin_seated
				NOT = {
					has_country_flag = ENG_cabinet_bevin_working
				}
				NOT = {
					has_country_flag = ENG_cabinet_bevin_lane_done
				}
			}
			ENG_cabinet_bevin_sync_tier = yes
			ENG_cabinet_bevin_start_next = yes
		}
	}
}

# Menzies's path: signals intelligence or special operations. The option not taken can never be completed.
country_event = {
	id = tfr_eng_cabinet.80
	title = tfr_eng_cabinet.80.t
	desc = tfr_eng_cabinet.80.d
	picture = GFX_report_event_generic_parliament
	is_triggered_only = yes
	option = {
		name = tfr_eng_cabinet.80.a
		ai_chance = {
			base = 60
		}
		custom_effect_tooltip = ENG_menzies_path_sigint_tt
		hidden_effect = {
			if = {
				limit = {
					NOT = {
						has_country_flag = ENG_menzies_path_sigint
					}
					NOT = {
						has_country_flag = ENG_menzies_path_soe
					}
				}
				set_country_flag = ENG_menzies_path_sigint
				clr_country_flag = ENG_cabinet_menzies_choice_pending
			}
		}
	}
	option = {
		name = tfr_eng_cabinet.80.b
		ai_chance = {
			base = 40
		}
		custom_effect_tooltip = ENG_menzies_path_soe_tt
		hidden_effect = {
			if = {
				limit = {
					NOT = {
						has_country_flag = ENG_menzies_path_sigint
					}
					NOT = {
						has_country_flag = ENG_menzies_path_soe
					}
				}
				set_country_flag = ENG_menzies_path_soe
				clr_country_flag = ENG_cabinet_menzies_choice_pending
			}
		}
	}
}
