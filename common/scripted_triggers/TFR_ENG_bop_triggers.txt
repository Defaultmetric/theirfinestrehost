### TFR - United Kingdom: Whitehall Balance triggers (part bop)
### Range based (explicit OR of ranges), so the centre range counts for neither side.

ENG_bop_appeasers_side_active = {
	custom_trigger_tooltip = {
		tooltip = ENG_bop_appeasers_side_active_tt
		OR = {
			is_power_balance_in_range = { id = ENG_whitehall_balance range = ENG_bop_appeasers_low_range }
			is_power_balance_in_range = { id = ENG_whitehall_balance range = ENG_bop_appeasers_mid_range }
			is_power_balance_in_range = { id = ENG_whitehall_balance range = ENG_bop_appeasers_high_range }
		}
	}
}
