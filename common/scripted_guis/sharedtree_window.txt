scripted_gui = {

	open_shared_tree_button = {
		context_type = player_context

		window_name = "shared_tree_window"
		parent_window_name = nationalfocusview
		
		effects = {
			open_shared_tree_button_click={
				if = {
					limit = {ROOT  = {NOT = {has_country_flag = shared_tree_clicked}}}
					ROOT = { set_country_flag = shared_tree_clicked mark_focus_tree_layout_dirty = yes }
				}
				else = 
				{
					ROOT = { clr_country_flag = shared_tree_clicked mark_focus_tree_layout_dirty = yes }
				}
			}
		}

		triggers = {
			open_shared_tree_button_click = {
				always = yes
			}
		}
	}

}
